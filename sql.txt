-- User Management System: exact SQL templates used by the application.
-- Application values are bound to ? by mysql2/promise execute(). Do not replace placeholders by concatenating user input.
-- Safe initialization (same as database/schema.sql):
CREATE DATABASE IF NOT EXISTS web_app CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE web_app;
CREATE TABLE IF NOT EXISTS Users (
  username VARCHAR(50) PRIMARY KEY,
  password_hash VARCHAR(255) NOT NULL,
  firstname VARCHAR(50) NOT NULL,
  lastname VARCHAR(50) NOT NULL,
  salary DECIMAL(12,2) NOT NULL,
  age INT NOT NULL,
  registerday DATE NOT NULL,
  signintime DATETIME NULL,
  INDEX idx_users_registerday (registerday),
  INDEX idx_users_names (firstname, lastname),
  INDEX idx_users_salary (salary),
  INDEX idx_users_age (age),
  INDEX idx_users_signintime (signintime)
);
-- No destructive reset statement is included. Schema reexecution leaves rows intact.
-- Health (query() call):
SELECT 1;
-- 1 Register: bindings [userid, bcrypt hash, first name, last name, salary, age].
INSERT INTO Users (username, password_hash, firstname, lastname, salary, age, registerday) VALUES (?, ?, ?, ?, ?, ?, CURDATE());
-- 2 Login: lookup [userid], bcrypt.compare in Node, then UPDATE [userid] only on success.
SELECT username, password_hash FROM Users WHERE username = ?;
UPDATE Users SET signintime = NOW() WHERE username = ?;
-- 3 Name search: the service chooses one of these SQL templates based on provided fields.
-- Values are escaped for literal LIKE matching: backslash, percent, underscore; then wrapped in percent signs.
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users WHERE firstname LIKE ? ESCAPE '\\' ORDER BY registerday ASC, username ASC LIMIT 200;
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users WHERE lastname LIKE ? ESCAPE '\\' ORDER BY registerday ASC, username ASC LIMIT 200;
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users WHERE firstname LIKE ? ESCAPE '\\' AND lastname LIKE ? ESCAPE '\\' ORDER BY registerday ASC, username ASC LIMIT 200;
-- 4 Exact userid [userid].
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users WHERE username = ? ORDER BY registerday ASC, username ASC LIMIT 200;
-- 5 Salary [minimum, maximum], inclusive.
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users WHERE salary BETWEEN ? AND ? ORDER BY registerday ASC, username ASC LIMIT 200;
-- 6 Age [minimum, maximum], inclusive.
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users WHERE age BETWEEN ? AND ? ORDER BY registerday ASC, username ASC LIMIT 200;
-- 7 Reference [userid], then [reference registerday]. Strictly later DATE; same-day rows excluded.
SELECT registerday FROM Users WHERE username = ?;
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users WHERE registerday > ? ORDER BY registerday ASC, username ASC LIMIT 200;
-- 8 Never signed in.
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users WHERE signintime IS NULL ORDER BY registerday ASC, username ASC LIMIT 200;
-- 9 Reference lookup above, then [reference registerday]. Includes the reference user.
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users WHERE registerday = ? ORDER BY registerday ASC, username ASC LIMIT 200;
-- 10 Today in database session timezone.
SELECT username, firstname, lastname, salary, age, registerday, signintime FROM Users WHERE registerday = CURDATE() ORDER BY registerday ASC, username ASC LIMIT 200;
-- Seed guard: bindings [john, maya, nina, omar, liam, zoe]. Seed aborts if ANY exists.
SELECT username FROM Users WHERE username IN (?, ?, ?, ?, ?, ?);
-- The seed calls beginTransaction() before its inserts, then commit() after all six.
START TRANSACTION;
-- Seed insert: bindings [userid, bcrypt hash, firstname, lastname, salary, age, offset days, sign-in offset or null, same sign-in offset].
-- Executed six times inside a transaction. No existing row is overwritten.
INSERT INTO Users (username, password_hash, firstname, lastname, salary, age, registerday, signintime) VALUES (?, ?, ?, ?, ?, ?, DATE_ADD(CURDATE(), INTERVAL ? DAY), CASE WHEN ? IS NULL THEN NULL ELSE DATE_ADD(DATE_ADD(CURDATE(), INTERVAL ? DAY), INTERVAL 9 HOUR) END);
COMMIT;
-- On insert failure the seed calls rollback() instead of commit(): ROLLBACK;
